javascriptconst axios = require('axios');

async function fetchAIPrediction() {
    try {
        const response = await axios.post('https://cheeseblockchain.com/api/developer/predict', {
            modelName: 'SmartContractAnalyzer',
            inputData: {
                code: 'contract Wallet { function transfer() public { msg.sender.call.value(1 ether)(); } }'
            }
        }, {
            headers: { 'x-api-key': 'YOUR_API_KEY' }
        });
        
        console.log('AI Audit Result:', response.data.prediction);
    } catch (err) {
        console.error('API Error:', err.message);
    }
}
